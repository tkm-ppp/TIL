## 3.コントローラーに近くの図書館を調べる処理を記述
```
def nearby
  if params[:lat].present? && params[:lng].present?
    user_lat = params[:lat].to_f
    user_lng = params[:lng].to_f

    # 全図書館から距離計算（Haversine公式を使用）
    @libraries = Library.all.map do |library|
      distance = calculate_distance(user_lat, user_lng, library.latitude, library.longitude)
      [library, distance]
    end

    # 距離が近い順にソート（30km以内に限定）
    @libraries = @libraries.select { |_, d| d <= 30 }.sort_by { |_, d| d }

    # 市ごとにグループ化
    @grouped_libraries = @libraries.group_by { |lib, _| lib.city }
  end
end

private

def calculate_distance(lat1, lng1, lat2, lng2)
  # ハーバーサイン公式による距離計算
  rad_per_deg = Math::PI/180
  dlat = lat2 - lat1
  dlng = lng2 - lng1

  a = Math.sin(dlat/2 * rad_per_deg)**2 + 
      Math.cos(lat1 * rad_per_deg) * 
      Math.cos(lat2 * rad_per_deg) * 
      Math.sin(dlng/2 * rad_per_deg)**2
  c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a))
  
  (6371 * c).round(2) # 地球の半径を6371kmで計算
end

```
