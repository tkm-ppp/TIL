# メソッドを渡す
name = :meth  # シンボルとしてメソッド名を格納

class Sample
  def meth
    puts "terminated"
  end
end

sample = Sample.new
sample.send(name)  # :meth というメソッドを呼び出す

# 同じになる
sample1 = :meth
sample2 = :meth
sample3 = 1
sample4 = 1

puts sample1 == sample2
puts sample3 == sample4
